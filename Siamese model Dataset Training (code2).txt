import tensorflow.keras.backend as K
import matplotlib.pyplot as plt
from sklearn.metrics import accuracy_score
from google.colab import drive

# ✅ Define dataset path
BASE_PATH = "/content/drive/MyDrive/facedectection.v2i.coco2"
TRAIN_FOLDER = os.path.join(BASE_PATH, "train")
VALID_FOLDER = os.path.join(BASE_PATH, "valid")
TEST_FOLDER = os.path.join(BASE_PATH, "test")

# ✅ Ensure dataset folders exist

    if not os.path.exists(folder):
        raise FileNotFoundError(f"🚨 ERROR: Folder '{folder}' does NOT exist!")

# ✅ Organize dataset into class subfolders (if images are misplaced)
def organize_images_into_class_folders(base_path):
    for folder in ["train", "valid", "test"]:
        folder_path = os.path.join(base_path, folder)
        class_folder = os.path.join(folder_path, "unknown")  # Create "unknown" class if none exist
        if not os.path.exists(class_folder):
            os.makedirs(class_folder)

        for file in os.listdir(folder_path):
            file_path = os.path.join(folder_path, file)
            if os.path.isfile(file_path):  # Move misplaced images
                shutil.move(file_path, os.path.join(class_folder, file))

organize_images_into_class_folders(BASE_PATH)

# ✅ Load and preprocess images
def load_image(image_path, target_size=(32, 32)):
    image = cv2.imread(image_path)
    if image is None:
        print(f"⚠️ Warning: Could not load image {image_path}")
        return np.zeros((target_size[0], target_size[1], 3))  # Return blank image
    image = cv2.resize(image, target_size)
    return image.astype("float32") / 255.0  # Normalize

# ✅ Create pairs for Siamese training
def create_pairs(image_folder):
    image_pairs, labels = [], []
    class_dict = {}

    for class_name in os.listdir(image_folder):
        class_path = os.path.join(image_folder, class_name)
        if os.path.isdir(class_path):
            images_in_class = [os.path.join(class_path, img) for img in os.listdir(class_path) if img.lower().endswith(('.png', '.jpg', '.jpeg'))]
            if len(images_in_class) < 2:  # Skip classes with <2 images
                continue
            class_dict[class_name] = images_in_class

    # 🔹 Generate pairs
    for class_name, image_paths in class_dict.items():
        # Positive pairs (same class)
        for i in range(len(image_paths)):
            for j in range(i + 1, len(image_paths)):
                image_pairs.append([image_paths[i], image_paths[j]])
                labels.append(1)

        # Negative pairs (different classes)
        other_classes = list(class_dict.keys())
        other_classes.remove(class_name)
        for other_class in other_classes:
            if class_dict.get(other_class):
                other_image = random.choice(class_dict[other_class])
                image_pairs.append([image_paths[0], other_image])
                labels.append(0)

    return np.array(image_pairs), np.array(labels)

# ✅ Generate training, validation, and test pairs
train_pairs, train_labels = create_pairs(TRAIN_FOLDER)
valid_pairs, valid_labels = create_pairs(VALID_FOLDER)
test_pairs, test_labels = create_pairs(TEST_FOLDER)

# ✅ Check generated pairs
print(f"Train pairs: {len(train_pairs)}")
print(f"Validation pairs: {len(valid_pairs)}")
print(f"Test pairs: {len(test_pairs)}")

# ✅ Create a custom generator
class ImagePairGenerator(tf.keras.utils.Sequence):
    def __init__(self, image_pairs, labels, batch_size=8, target_size=(32, 32)):
        self.image_pairs = image_pairs
        self.labels = labels
        self.batch_size = batch_size
        self.target_size = target_size

    def __len__(self):
        return int(np.floor(len(self.image_pairs) / self.batch_size))

    def __getitem__(self, index):
        batch_pairs = self.image_pairs[index * self.batch_size:(index + 1) * self.batch_size]
        batch_labels = self.labels[index * self.batch_size:(index + 1) * self.batch_size]

        image_1 = np.array([load_image(pair[0], self.target_size) for pair in batch_pairs])
        image_2 = np.array([load_image(pair[1], self.target_size) for pair in batch_pairs])

        return (image_1, image_2), np.array(batch_labels)

# ✅ Initialize data generators
train_generator = ImagePairGenerator(train_pairs, train_labels, batch_size=2048)
valid_generator = ImagePairGenerator(valid_pairs, valid_labels, batch_size=2048)

# ✅ Build Siamese Model
def build_siamese_model(input_shape):
    input_image = Input(shape=input_shape)
    x = Conv2D(64, (3, 3), activation="relu", padding="same")(input_image)
    x = Conv2D(128, (3, 3), activation="relu", padding="same")(x)
    x = Flatten()(x)
    x = Dense(128, activation="relu")(x)
    return Model(inputs=input_image, outputs=x)

# ✅ Euclidean distance function
def euclidean_distance(vectors):
    x, y = vectors
    sum_square = K.sum(K.square(x - y), axis=1, keepdims=True)
    return K.sqrt(K.maximum(sum_square, K.epsilon()))

# ✅ Define input shape
INPUT_SHAPE = (32, 32, 3)

# ✅ Create base network
base_model = build_siamese_model(INPUT_SHAPE)

# ✅ Define inputs for the two images
input_image_1 = Input(shape=INPUT_SHAPE)
input_image_2 = Input(shape=INPUT_SHAPE)

# ✅ Process both images through the base network
processed_image_1 = base_model(input_image_1)
processed_image_2 = base_model(input_image_2)

# ✅ Compute Euclidean distance
distance = Lambda(euclidean_distance)([processed_image_1, processed_image_2])

# ✅ Output layer
output = Dense(1, activation="sigmoid")(distance)

# ✅ Create Siamese model
siamese_model = Model(inputs=[input_image_1, input_image_2], outputs=output)

# ✅ Compile model
siamese_model.compile(optimizer="adam", loss="binary_crossentropy", metrics=["accuracy"])

# ✅ Train model
history = siamese_model.fit(
    train_generator,
    validation_data=valid_generator,
    epochs=10
)

# ✅ Save model
siamese_model.save_weights("siamese_model_weights.h5")

# ✅ Evaluate on test data
test_images_1 = np.array([load_image(pair[0], target_size=(32, 32)) for pair in test_pairs])
test_images_2 = np.array([load_image(pair[1], target_size=(32, 32)) for pair in test_pairs])

test_predictions = siamese_model.predict([test_images_1, test_images_2])
test_predictions_binary = (test_predictions > 0.5).astype(int)

accuracy = accuracy_score(test_labels, test_predictions_binary)
print(f"✅ Test Accuracy: {accuracy * 100:.2f}%")
